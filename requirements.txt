# Upper bounds on the two that have broken this project with a major release.
# sentence-transformers 2.2.2 against a huggingface_hub that had dropped
# cached_download; langchain 1.0 moving langchain.chains to langchain-classic.
# Both were only noticed from a clean install. Dependabot proposes the bumps;
# the image build and the dimension check in CI judge them.
sentence-transformers>=3.0,<7
langchain>=1.0,<2
# create_retrieval_chain and create_stuff_documents_chain moved here in
# langchain 1.0; it is a dependency of langchain but listed because we import it.
langchain-classic>=1.0,<2
# Imported directly for ChatPromptTemplate; it arrives transitively via
# langchain too, but a transitive dependency is not a declared one.
langchain-core>=1.0,<2
flask
# Imported for SecretStr; transitive via langchain, but we import it.
pydantic
pypdf
python-dotenv
# The grpc extra only exists up to a point: pip warns that newer pinecone
# releases do not provide it and backtracks to one that does. store_index.py
# imports pinecone.grpc, so the extra has to resolve.
pinecone[grpc]
# Lower bounds matter more here than upper ones. Without this, resolving
# pinecone[grpc] made pip walk langchain-pinecone backwards to 0.0.1 -- a 2023
# release with no PineconeVectorStore -- and the image installed it silently.
langchain-pinecone>=0.2,<1
langchain-community>=0.3,<1
langchain-groq>=1.0,<2
gunicorn
-e .
